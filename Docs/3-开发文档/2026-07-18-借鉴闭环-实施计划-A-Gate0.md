$baselineRoot = Join-Path $env:TEMP 'czxt-borrowing-prework-baseline'
$manifestPath = Join-Path $baselineRoot 'baseline-manifest.json'
$manifest = Get-Content $manifestPath -Raw -Encoding UTF8 | ConvertFrom-Json
if ($manifest.Schema -ne 'czxt/baseline-manifest/v1') { throw '基线 manifest schema 错误' }
if ([IO.Path]::GetFullPath($manifest.WorkspaceRoot).TrimEnd('\') -ne [IO.Path]::GetFullPath($PWD.Path).TrimEnd('\')) { throw '基线工作区不匹配' }
if ($manifest.Head -ne (git rev-parse HEAD)) { throw '基线 HEAD 不匹配' }
foreach ($artifact in @($manifest.Artifacts)) {
  $path = Join-Path $baselineRoot $artifact.Path
  if (-not (Test-Path -LiteralPath $path -PathType Leaf)) { throw "基线缺产物：$($artifact.Path)" }
  if ((Get-Item -LiteralPath $path).Length -ne [int64]$artifact.Length -or
      (Get-FileHash -LiteralPath $path -Algorithm SHA256).Hash -ne $artifact.Sha256) {
    throw "基线产物被替换：$($artifact.Path)"
  }
}

$trackedDoc = Get-Content (Join-Path $baselineRoot 'winps-decoded-text-baseline.json') -Raw -Encoding UTF8 | ConvertFrom-Json
$trackedExpected = @($trackedDoc.Rows)
if ($trackedDoc.Schema -ne 'czxt/winps-decoded-text-baseline/v1' -or
    [int]$trackedDoc.Count -ne 124 -or $trackedExpected.Count -ne 124) { throw '原始 PS1 基线身份或行数错误' }

$gate = Join-Path $PWD '能力资产/tools/scripts/check-winps-encoding.ps1'
$allFrameworkPs1 = @(& powershell -NoProfile -ExecutionPolicy Bypass -File $gate -Root $PWD -ListOnly)
if ($LASTEXITCODE -ne 0 -or $allFrameworkPs1.Count -eq 0) { throw '无法取得 framework PS1 清单' }
if (@($allFrameworkPs1 | Where-Object { [string]::IsNullOrWhiteSpace($_) }).Count) { throw '清单含空行' }

New-Item -ItemType Directory -Path $migrationRoot | Out-Null

function Get-DecodedRows([string[]]$RelativePaths) {
  $strictUtf8 = [Text.UTF8Encoding]::new($false, $true)
  foreach ($rel in $RelativePaths) {
    $text = [IO.File]::ReadAllText((Join-Path $PWD $rel), $strictUtf8)
    $bytes = [Text.UTF8Encoding]::new($false).GetBytes($text)
    $sha = [Security.Cryptography.SHA256]::Create().ComputeHash($bytes)
    [pscustomobject]@{ Path=$rel; TextSha256=([BitConverter]::ToString($sha)-replace '-',''); CharacterLength=$text.Length }
  }
}

$allScopeBaseline = @(Get-DecodedRows $allFrameworkPs1)
$scopeDoc = [ordered]@{

```powershell
function Assert-SameRows($Expected, $Actual, [string]$Label) {
  $expectedMap = @{}; foreach ($row in @($Expected)) { $expectedMap[$row.Path] = $row }
  $actualMap = @{}; foreach ($row in @($Actual)) { $actualMap[$row.Path] = $row }
  if ($expectedMap.Count -ne $actualMap.Count) { throw "$Label 文件数变化" }
  foreach ($path in $expectedMap.Keys) {
    if (-not $actualMap.ContainsKey($path)) { throw "$Label 缺文件：$path" }
    if ($expectedMap[$path].TextSha256 -ne $actualMap[$path].TextSha256 -or
        [int]$expectedMap[$path].CharacterLength -ne [int]$actualMap[$path].CharacterLength) {
      throw "$Label 解码内容变化：$path"
    }
  }
}

$trackedActual = @(Get-DecodedRows @(git ls-files '*.ps1'))
$scopeDoc = Get-Content (Join-Path $migrationRoot 'all-scope-decoded-text-baseline.json') -Raw -Encoding UTF8 | ConvertFrom-Json
$scopeExpected = @($scopeDoc.Rows)
if ($scopeDoc.Schema -ne 'czxt/winps-migration-scope-baseline/v1' -or
    [int]$scopeDoc.Count -ne $allFrameworkPs1.Count -or
    $scopeExpected.Count -ne $allFrameworkPs1.Count) { throw '正式范围基线身份或行数异常' }
$scopeActual = @(Get-DecodedRows $allFrameworkPs1)
foreach ($row in @($trackedExpected + $scopeExpected)) {
  if ([string]::IsNullOrWhiteSpace([string]$row.Path) -or
      [string]::IsNullOrWhiteSpace([string]$row.TextSha256) -or
      $null -eq $row.CharacterLength) { throw '基线存在缺字段行' }
}
if (@($trackedExpected.Path | Sort-Object -Unique).Count -ne 124) { throw '原始跟踪 PS1 基线路径重复' }
if (@($scopeExpected.Path | Sort-Object -Unique).Count -ne $scopeExpected.Count) { throw '正式范围基线路径重复' }
Assert-SameRows $trackedExpected $trackedActual '原始 124 个跟踪 PS1'
Assert-SameRows $scopeExpected $scopeActual '正式范围 PS1'
```

function Get-CzxtRootMode {
  param([string]$Root)
  $template = Test-Path -LiteralPath (Join-Path $Root '.czxt-template-root') -PathType Leaf
  $project = Test-Path -LiteralPath (Join-Path $Root '.czxt-project-root') -PathType Leaf

}
function Test-IsTemplateRoot {

}
